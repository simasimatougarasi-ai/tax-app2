    let taxChartInstance = null;


    // 入力するたびに年額を自動計算して表示する機能

    function calculateLiveTotals() {

        const monthlyIncome = parseInt(document.getElementById('monthlyIncome').value) || 0;

        document.getElementById('liveAnnualIncome').textContent = (monthlyIncome * 12).toLocaleString();


        const expenses = ['rent', 'utilities', 'telecom', 'transport', 'food', 'social', 'hobby', 'other'];

        let monthlyExpenses = 0;

        expenses.forEach(id => {

            monthlyExpenses += parseInt(document.getElementById(id).value) || 0;

        });

        document.getElementById('liveAnnualExpenses').textContent = (monthlyExpenses * 12).toLocaleString();

    }


    function calculateTaxes() {

        const monthlyIncome = parseInt(document.getElementById('monthlyIncome').value) || 0;

        const income = monthlyIncome * 12;


        if (income === 0) {

            alert('月収を入力してください。');

            return;

        }


        // 月々の入力値を取得

        const rent = parseInt(document.getElementById('rent').value) || 0;

        const utilities = parseInt(document.getElementById('utilities').value) || 0;

        const telecom = parseInt(document.getElementById('telecom').value) || 0;

        const transport = parseInt(document.getElementById('transport').value) || 0;

        const food = parseInt(document.getElementById('food').value) || 0;

        const social = parseInt(document.getElementById('social').value) || 0;

        const hobby = parseInt(document.getElementById('hobby').value) || 0;

        const other = parseInt(document.getElementById('other').value) || 0;

        const savings = parseInt(document.getElementById('savings').value) || 0;


        // 年間の計算

        const annualRent = rent * 12;

        const annualSavings = savings * 12;

        // 消費税がかかる生活費（家賃・貯金以外）

        const annualTaxableSpending = (utilities + telecom + transport + food + social + hobby + other) * 12;

        const annualSpending = annualRent + annualTaxableSpending;


        // 1. 社会保険料の概算 (約15%)

        const socialInsurance = Math.floor(income * 0.15);


        // 2. 給与所得控除の概算計算

        let incomeDeduction = 0;

        if (income <= 1625000) incomeDeduction = 550000;

        else if (income <= 1800000) incomeDeduction = income * 0.4 - 100000;

        else if (income <= 3600000) incomeDeduction = income * 0.3 + 80000;

        else if (income <= 6600000) incomeDeduction = income * 0.2 + 440000;

        else if (income <= 8500000) incomeDeduction = income * 0.1 + 1100000;

        else incomeDeduction = 1950000;


        // 3. 課税所得の概算 (基礎控除を所得税48万、住民税43万とする)

        let taxableIncomeTax = income - socialInsurance - incomeDeduction - 480000;

        if (taxableIncomeTax < 0) taxableIncomeTax = 0;

        

        let taxableResidentTax = income - socialInsurance - incomeDeduction - 430000;

        if (taxableResidentTax < 0) taxableResidentTax = 0;


        // 4. 所得税の概算

        let incomeTax = 0;

        if (taxableIncomeTax <= 1949000) incomeTax = taxableIncomeTax * 0.05;

        else if (taxableIncomeTax <= 3299000) incomeTax = taxableIncomeTax * 0.10 - 97500;

        else if (taxableIncomeTax <= 6949000) incomeTax = taxableIncomeTax * 0.20 - 427500;

        else if (taxableIncomeTax <= 8999000) incomeTax = taxableIncomeTax * 0.23 - 636000;

        else if (taxableIncomeTax <= 17999000) incomeTax = taxableIncomeTax * 0.33 - 1536000;

        else incomeTax = taxableIncomeTax * 0.40 - 2796000;

        incomeTax = Math.floor(incomeTax);


        // 5. 住民税の概算 (約10% + 均等割5000円)

        const residentTax = taxableResidentTax > 0 ? Math.floor(taxableResidentTax * 0.10) + 5000 : 0;


        // 6. 消費税の概算 (家賃と貯金は非課税として除外し、10%計算)

        const consumptionTax = Math.floor(annualTaxableSpending * (10 / 110));

        const pureSpending = annualSpending - consumptionTax; // 税抜の総支出（家賃含む）


        // 7. 計算上の余り（収入から各種税金・支出・貯金を引いた残り）

        let remainingMoney = income - socialInsurance - incomeTax - residentTax - annualSpending - annualSavings;

        if (remainingMoney < 0) remainingMoney = 0; // マイナス（赤字）の場合は0表示


        // 総負担額

        const totalTaxes = socialInsurance + incomeTax + residentTax + consumptionTax;


        // UIの更新

        updateUI(income, socialInsurance, incomeTax, residentTax, consumptionTax, pureSpending, annualSavings, remainingMoney, totalTaxes);

        drawChart(socialInsurance, incomeTax, residentTax, consumptionTax, pureSpending, annualSavings, remainingMoney);

    }


    function updateUI(income, soc, incTax, resTax, consTax, spend, savings, remain, total) {

        document.getElementById('resultSection').style.display = 'flex';

        

        const detailsHtml = `

            <h3>内訳（年間概算）</h3>

            <div class="detail-item"><span>額面年収:</span> <span>${income.toLocaleString()} 円</span></div>

            <div class="detail-item"><span>社会保険料:</span> <span>${soc.toLocaleString()} 円</span></div>

            <div class="detail-item"><span>所得税:</span> <span>${incTax.toLocaleString()} 円</span></div>

            <div class="detail-item"><span>住民税:</span> <span>${resTax.toLocaleString()} 円</span></div>

            <div class="detail-item"><span>消費税(家賃以外から算出):</span> <span>${consTax.toLocaleString()} 円</span></div>

            <div class="detail-item"><span>その他の支出(税抜・家賃含):</span> <span>${spend.toLocaleString()} 円</span></div>

            <div class="detail-item"><span>入力した貯金額:</span> <span>${savings.toLocaleString()} 円</span></div>

            <div class="detail-item"><span>未割り当ての余り:</span> <span>${remain.toLocaleString()} 円</span></div>

            <div class="detail-item total"><span>国・自治体に納めた総額:</span> <span>${total.toLocaleString()} 円</span></div>

        `;

        document.getElementById('detailsList').innerHTML = detailsHtml;

    }


    function drawChart(soc, incTax, resTax, consTax, spend, savings, remain) {

        const ctx = document.getElementById('taxChart').getContext('2d');

        

        if (taxChartInstance) {

            taxChartInstance.destroy();

        }


        taxChartInstance = new Chart(ctx, {

            type: 'pie',

            data: {

                labels: ['社会保険料', '所得税', '住民税', '消費税', '生活費(税抜)', '貯金', '未割り当て'],

                datasets: [{

                    data: [soc, incTax, resTax, consTax, spend, savings, remain],

                    backgroundColor: [

                        '#3498db', // 水色: 社会保険（指定）

                        '#f39c12', // オレンジ: 所得税

                        '#f1c40f', // 黄色: 住民税

                        '#e74c3c', // 赤色: 消費税（指定）

                        '#95a5a6', // 灰: 支出

                        '#2ecc71', // 緑: 貯金

                        '#ecf0f1'  // 薄灰: 余り

                    ],

                    borderWidth: 1

                }]

            },

            options: {

                responsive: true,

                maintainAspectRatio: false,

                plugins: {

                    legend: {

                        position: 'bottom',

                    },

                    tooltip: {

                        callbacks: {

                            label: function(context) {

                                let label = context.label || '';

                                if (label) {

                                    label += ': ';

                                }

                                if (context.parsed !== null) {

                                    label += new Intl.NumberFormat('ja-JP').format(context.parsed) + ' 円';

                                }

                                return label;

                            }

                        }

                    }

                }

            }

        });

    }
